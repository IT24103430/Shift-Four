using System;
using System.Collections.Generic;
using System.Linq;
using NUnit.Framework;
using UnityEditor;
using UnityEngine;
using Object = UnityEngine.Object;

namespace ShiftFour.Tests
{
    public sealed class TacticalArmsSkinWeightEditModeTests
    {
        private GameObject instance;
        private SkinnedMeshRenderer skin;
        private Mesh baked;
        private Vector3[] rest;
        private int[] dorsal;
        private (int, int)[] edges;

        [SetUp] public void Setup()
        {
            instance = Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Models/FPS/SF_TacticalArms.fbx"));
            skin = instance.GetComponentInChildren<SkinnedMeshRenderer>();
            baked = new Mesh();
            skin.BakeMesh(baked, true); rest = baked.vertices;
            int region = Array.FindIndex(skin.sharedMaterials, m => m.name == "GloveReinforcement");
            Assert.That(region, Is.GreaterThanOrEqualTo(0));
            dorsal = skin.sharedMesh.GetTriangles(region).Distinct().ToArray();
            var unique = new HashSet<(int, int)>();
            int[] triangles = skin.sharedMesh.triangles;
            for (int t = 0; t < triangles.Length; t += 3)
                for (int j = 0; j < 3; j++)
                {
                    int a = triangles[t + j], b = triangles[t + (j + 1) % 3];
                    unique.Add(a < b ? (a, b) : (b, a));
                }
            edges = unique.ToArray();
        }

        [TearDown] public void Cleanup() { Object.DestroyImmediate(baked); Object.DestroyImmediate(instance); }

        [Test] public void DorsalPanels_AreNotDrivenByFingerOrThumbJoints()
        {
            BoneWeight[] weights = skin.sharedMesh.boneWeights;
            foreach (int index in dorsal)
                foreach (var influence in Influences(weights[index]))
                    if (influence.weight > .000001f)
                        Assert.That(skin.bones[influence.bone].name, Does.StartWith("hand.").Or.StartWith("palm_"), "Dorsal vertex " + index);
        }

        [Test] public void Cuffs_FollowForearmsWithoutFingerInfluences()
        {
            int region = Array.FindIndex(skin.sharedMaterials, m => m.name == "CuffElastic");
            Assert.That(region, Is.GreaterThanOrEqualTo(0));
            BoneWeight[] weights = skin.sharedMesh.boneWeights;
            foreach (int index in skin.sharedMesh.GetTriangles(region).Distinct())
                foreach (var influence in Influences(weights[index]))
                    if (influence.weight > .000001f)
                        Assert.That(skin.bones[influence.bone].name, Does.StartWith("forearm."), "Cuff vertex " + index);
        }

        [TestCase("index", 75, 80, 40)]
        [TestCase("middle", 75, 80, 40)]
        [TestCase("ring", 75, 80, 40)]
        [TestCase("pinky", 75, 80, 40)]
        [TestCase("index", -25, -20, -15)]
        [TestCase("middle", -25, -20, -15)]
        [TestCase("ring", -25, -20, -15)]
        [TestCase("pinky", -25, -20, -15)]
        public void IndividualFingerStress_DoesNotStretchDorsalPanels(string finger, float a, float b, float c)
        {
            var rotations = skin.bones.Select(bone => bone.localRotation).ToArray();
            foreach (string side in new[] { "L", "R" })
                foreach (var segment in new[] { ("01", a), ("02", b), ("03", c) })
                {
                    Transform joint = skin.bones.Single(bone => bone.name == "f_" + finger + "." + segment.Item1 + "." + side);
                    joint.localRotation *= Quaternion.Euler(segment.Item2, 0, 0);
                }
            skin.BakeMesh(baked, true); Vector3[] moved = baked.vertices;
            Assert.That(moved.All(Finite), Is.True);
            Assert.That(dorsal.Max(i => Vector3.Distance(rest[i], moved[i])), Is.LessThan(.0001f), "Finger rotation must leave back-of-hand panels attached to the metacarpals.");
            // These compound rotations deliberately exceed the shipped pose.
            // Record whole-skin strain separately; this regression specifically
            // protects the dorsal reinforcement from unrelated finger joints.
            float growth = edges.Max(e => Vector3.Distance(moved[e.Item1], moved[e.Item2]) - Vector3.Distance(rest[e.Item1], rest[e.Item2]));
            TestContext.WriteLine("Whole-skin stress edge growth (m): " + growth.ToString("R"));
            for (int i = 0; i < rotations.Length; i++) skin.bones[i].localRotation = rotations[i];
            skin.BakeMesh(baked, true);
            Assert.That(rest.Zip(baked.vertices, Vector3.Distance).Max(), Is.LessThan(.0001f));
        }

        [TestCase("index", 15)] [TestCase("index", -15)]
        [TestCase("middle", 15)] [TestCase("middle", -15)]
        [TestCase("ring", 15)] [TestCase("ring", -15)]
        [TestCase("pinky", 15)] [TestCase("pinky", -15)]
        [TestCase("thumb", 15)] [TestCase("thumb", -15)]
        public void PresentationRangeFingerCurl_PreservesContinuousSkin(string finger, float angle)
        {
            foreach (string side in new[] { "L", "R" })
            {
                string name = (finger == "thumb" ? "thumb" : "f_" + finger) + ".01." + side;
                skin.bones.Single(b => b.name == name).localRotation *= Quaternion.Euler(angle, 0, 0);
            }
            skin.BakeMesh(baked, true); Vector3[] moved = baked.vertices;
            Assert.That(moved.All(Finite), Is.True);
            Assert.That(edges.Max(e => Vector3.Distance(moved[e.Item1], moved[e.Item2]) - Vector3.Distance(rest[e.Item1], rest[e.Item2])), Is.LessThan(.006f));
        }

        [TestCase("hand.L", 35, 0, 0)] [TestCase("hand.R", -35, 0, 0)]
        [TestCase("hand.L", 0, 0, 30)] [TestCase("forearm.L", 0, 75, 0)]
        public void WristAndReloadRange_PreserveFiniteContinuousSkin(string name, float x, float y, float z)
        {
            Transform bone = skin.bones.Single(b => b.name == name);
            bone.localRotation *= Quaternion.Euler(x, y, z);
            skin.BakeMesh(baked, true); Vector3[] moved = baked.vertices;
            Assert.That(moved.All(Finite), Is.True);
            Assert.That(edges.Max(e => Vector3.Distance(moved[e.Item1], moved[e.Item2]) - Vector3.Distance(rest[e.Item1], rest[e.Item2])), Is.LessThan(.006f));
        }

        private static IEnumerable<(int bone, float weight)> Influences(BoneWeight w)
        {
            yield return (w.boneIndex0, w.weight0); yield return (w.boneIndex1, w.weight1);
            yield return (w.boneIndex2, w.weight2); yield return (w.boneIndex3, w.weight3);
        }
        private static bool Finite(Vector3 p) => !(float.IsNaN(p.x) || float.IsInfinity(p.x) || float.IsNaN(p.y) || float.IsInfinity(p.y) || float.IsNaN(p.z) || float.IsInfinity(p.z));
    }
}
