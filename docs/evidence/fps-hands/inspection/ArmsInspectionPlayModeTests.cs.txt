  void Measure(string scenario,float p)
  {
   var mesh=new Mesh(); skin.BakeMesh(mesh,true);
   var v=mesh.vertices.Select(x=>camera.transform.InverseTransformPoint(skin.transform.TransformPoint(x))).ToArray();
   var tri=mesh.triangles; float near=camera.nearClipPlane; int crossings=0,visible=0;
   float yMax=near*Mathf.Tan(camera.fieldOfView*Mathf.Deg2Rad*.5f),xMax=yMax*camera.aspect;
   for(int i=0;i<tri.Length;i+=3)
   {
    Vector3[] a={v[tri[i]],v[tri[i+1]],v[tri[i+2]]};
    if(a.All(x=>x.z>=near)||a.All(x=>x.z<near))continue;
    crossings++; var pts=new System.Collections.Generic.List<Vector3>();
    for(int j=0;j<3;j++) {var x=a[j];var y=a[(j+1)%3];if((x.z<near)!=(y.z<near))pts.Add(Vector3.Lerp(x,y,(near-x.z)/(y.z-x.z)));}
    if(pts.Count>0&&!pts.All(x=>x.x< -xMax)&&!pts.All(x=>x.x>xMax)&&!pts.All(x=>x.y< -yMax)&&!pts.All(x=>x.y>yMax))visible++;
   }
   float gunZ=float.PositiveInfinity;
   foreach(var f in model.GetComponentsInChildren<MeshFilter>().Where(f=>f.name.Contains("FNFiveSeven_")&&f.GetComponent<Renderer>().enabled))
   { var b=f.sharedMesh.bounds;for(int i=0;i<8;i++){var x=b.center+Vector3.Scale(b.extents,new Vector3((i&1)==0?-1:1,(i&2)==0?-1:1,(i&4)==0?-1:1));gunZ=Mathf.Min(gunZ,camera.transform.InverseTransformPoint(f.transform.TransformPoint(x)).z);}}
   csv.AppendLine(string.Join(",",scenario,p,Get(mount,"PresentationRetreat"),camera.aspect,camera.fieldOfView,near,v.Min(x=>x.z),v.Max(x=>x.z),v.Count(x=>x.z<near),crossings,visible,gunZ,Vector3.Distance(Bone("forearm.L").position,Bone("hand.L").position),Vector3.Distance(Bone("hand.L").position,Bone("f_middle.01.L").position)));
   Object.Destroy(mesh);
  }
  IEnumerator Capture(string name,int width=1600,int height=900)
  {
   var rt=new RenderTexture(width,height,24); rt.Create(); camera.aspect=(float)width/height; camera.targetTexture=rt;

   var tex=new Texture2D(width,height,TextureFormat.RGB24,false);tex.ReadPixels(new Rect(0,0,width,height),0,0);tex.Apply();
   using(var writer=new BinaryWriter(File.Create(Out+"/"+name+".tga"))) {writer.Write(new byte[]{0,0,2,0,0,0,0,0,0,0,0,0});writer.Write((short)width);writer.Write((short)height);writer.Write((byte)24);writer.Write((byte)0);foreach(Color32 c in tex.GetPixels32()){writer.Write(c.b);writer.Write(c.g);writer.Write(c.r);}}
   RenderTexture.active=previous;camera.targetTexture=null;rt.Release();Object.Destroy(rt);Object.Destroy(tex);
  }
