using System;using System.IO;using System.Text.RegularExpressions;using UnityEditor;
[InitializeOnLoad]public static class QaGraphicalExit {
 static QaGraphicalExit(){if(!string.IsNullOrEmpty(Environment.GetEnvironmentVariable("SHIFTFOUR_GRAPHICAL_TEST_RESULT")))EditorApplication.update+=Tick;}
 static void Tick(){var p=Environment.GetEnvironmentVariable("SHIFTFOUR_GRAPHICAL_TEST_RESULT");if(EditorApplication.isPlayingOrWillChangePlaymode||!File.Exists(p))return;try{var s=File.ReadAllText(p);if(!s.TrimEnd().EndsWith("</test-run>"))return;var m=Regex.Match(s,"failed=\"([0-9]+)\"");EditorApplication.Exit(m.Success&&m.Groups[1].Value=="0"?0:2);}catch(IOException){}}
}
